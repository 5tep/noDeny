SELECT 
    grp_id AS ID,
    LEFT(grp_name, 20) AS MNEMONIC,
    '2008-01-02 00:00:00' AS BEGIN_TIME,
    '2099-12-31 23:59:59' AS END_TIME,
    grp_name AS DESCRIPTION,
    18 AS REGION_ID

OPTIONALLY ENCLOSED BY ''
LINES TERMINATED BY '\n'
FROM 
    user_grp;
" > /var/lib/mysql-files/query.sql
